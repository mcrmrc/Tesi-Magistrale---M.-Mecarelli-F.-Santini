\justifying
ICMP è un protocollo che opera al livello di rete e permette: 
\begin{itemize}
\item \textbf{Diagnostica della rete}  
\item \textbf{Segnalazione di errori}
\item \textbf{Risoluzione dei problemi}
\item \textbf{Ottenimento di informazioni} 
\end{itemize}

\begin{frame}
\frametitle{Tipologie di messaggi ICMP} 
\textbf{Messaggi di errore}
\begin{itemize}
\item Destination Unreachable (IPv4, IPv6)
\item Time Exceeded (IPv4, IPv6)
\item Parameter Problem (IPv4, IPv6)
\item Source Quench (IPv4)
\item Redirect (IPv4)
\item Packet Too Big (IPv6)
\end{itemize} 
\textbf{Messaggi di informazione}
\begin{itemize}
\item Echo Request/Reply (IPv4, IPv6)
\item Timestamp Request/Reply (IPv4)
\item Information Request/Reply (IPv4)
\end{itemize} 
\end{frame}
