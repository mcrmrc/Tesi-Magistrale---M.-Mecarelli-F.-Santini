\begin{frame}
Spunti ricavati dall'analisi di ICMP Tunnel e ICMP Door: 
\begin{itemize}
\item Presenza di proxy intermediari fra le due macchine
\item La quantità di dati che si invia
\item Le tipologie di messaggi utilizzati ed i campi utilizzati
\end{itemize}

\begin{frame} 
\frametitle{Flusso dell'entità attaccante (che richiede i dati)} 
\begin{enumerate}
\item Definisce: 
\begin{itemize} 
\item l'indirizzo IP della vittima 
\item la lista dei proxy da utilizzare
\item il Covert Channel per lo scambio dei dati
\end{itemize} 
\item Ricava i proxy disponibili
\item Invia delle informazioni
\item Aspetta di ricevere delle informazioni
\end{enumerate}
\end{frame}

\begin{frame} 
\frametitle{Flusso dell'entità proxy (che inoltra i dati)} 
\begin{enumerate}
\item Definisce: 
\begin{itemize} 
\item l'indirizzo IP dell'attaccante
\end{itemize} 
\item Stabilisce una connessione con le macchine di cui farà da tramite 
\item Aspetta di ricevere delle informazioni  da una macchina
\item Inoltra le informazioni ricevute dall'altra macchina
\end{enumerate}
\end{frame}

\begin{frame} 
\frametitle{Flusso dell'entità vittima (che invia i dati)} 
\begin{enumerate}
\item Definisce: 
\begin{itemize} 
\item minimo numero di entità connesse necessarie per l'invio dei dati.
\end{itemize} 
\item Attende che i proxy si connettano con essa
\item Attende di ricevere delle informazioni da una delle macchine connesse
\item Invia la conferma della ricezione dell'informazione o i risultati del comando ricevuto
\end{enumerate}
\end{frame}
