\begin{frame}
\frametitle{Covert Channel} 
Minaccia che permette la capacità di comunicare o trasferire dati in maniera non autorizzata.
\vspace{1ex} \newline  
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth} 
%\justifying
\textbf{Caratteristiche}:  
\begin{itemize}
\item Furtività
\item Capacità di trasmissione 
%\item Uso delle risorse
%\item Rumore
\item Indistinguibilità
\end{itemize} 
\column{0.48\textwidth} 
\textbf{Tipologie}: 
\begin{itemize}
\item Timing Covert Channel 
\item Storage Covert Channel
\item Behavioural Covert Channel
\end{itemize} 
\end{columns}
\end{frame}
